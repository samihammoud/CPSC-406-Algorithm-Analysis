\subsubsection*{Product automata}

We want to construct an automaton that recognizes exactly those words that are in both $L(\mathcal{A}^{(1)})$ and $L(\mathcal{A}^{(2)})$ by keeping track of both automata simultaneously.

\begin{definition}[Product automaton]
For DFAs $\mathcal{A}^{(1)} = (Q^{(1)}, \Sigma, \delta^{(1)}, q_0^{(1)}, F^{(1)})$ and $\mathcal{A}^{(2)} = (Q^{(2)}, \Sigma, \delta^{(2)}, q_0^{(2)}, F^{(2)})$ we define $\mathcal{A}=(Q, \Sigma, \delta, q_0, F)$ by:
\begin{itemize}
  \item $Q := Q^{(1)} \times Q^{(2)}$ (states are pairs $(q^{(1)}, q^{(2)})$)
  \item $\delta((q^{(1)}, q^{(2)}), a) := (\delta^{(1)}(q^{(1)}, a), \delta^{(2)}(q^{(2)}, a))$
  \item $q_0 := (q_0^{(1)}, q_0^{(2)})$
\end{itemize}
The set $F$ is chosen depending on the operation (intersection, union, etc.).
\end{definition}

\begin{definition}[Intersection automaton]
With notation as above, the \textbf{intersection automaton} is the product automaton with $F := F^{(1)} \times F^{(2)}$. Then $L(\mathcal{A}) = L(\mathcal{A}^{(1)}) \cap L(\mathcal{A}^{(2)})$.
\end{definition}
